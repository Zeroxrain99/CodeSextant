### 2.1 期許 3「一開始視野要近」+ 期許 2「連線明顯」→ 開場 frameNodes 飛入

function pickFocusNode() {  // 改用 degree（連線最密）為主，非 rank
  let bi = 0, bd = -1;

**framing 算法（fov 三角，R3 同社群優先 + 刪 K 截斷 + 刪自距離 bug）**：

```js
const FRAME_FILL = 0.65;   // 焦點+鄰居佔視野 65%（業界 0.6~0.8），graph_frame_fill 可調
function frameNodes(indices, durMs) {
  const ctr = new THREE.Vector3();
  indices.forEach(i => ctr.add(nodeWorld[i]));
  ctr.multiplyScalar(1 / indices.length);
  let r = 60;   // 保底（單節點/極近防 r=0 貼臉）
  indices.forEach(i => { r = Math.max(r, ctr.distanceTo(nodeWorld[i])); });  // 刪除草案誘餌自距離行
  const fovV = camera.fov * Math.PI / 180;
  const fovH = 2 * Math.atan(Math.tan(fovV / 2) * camera.aspect);
  const dist = Math.max(r / FRAME_FILL / Math.tan(fovV / 2), r / FRAME_FILL / Math.tan(fovH / 2));
  // R3 同社群優先已在呼叫端過濾鄰居；RAW-06：方向預設保留當前視角（graph_frame_reorient=off）
  const dir = camera.position.clone().sub(controls.target).normalize();
  startFly(ctr.clone().add(dir.multiplyScalar(dist)), ctr.clone(), durMs);
}
// 鄰居選擇：同社群優先（R3），框全部 1-hop（刪 K 截斷，real≤119/synth≤18 不會拉成全景）
function focusNeighbors(fi) {
  const c = graph.nodes[fi].community;
  const same = adj[fi].filter(j => graph.nodes[j].community === c);
  return same.length >= 3 ? [fi, ...same] : [fi, ...adj[fi]];   // 同社群太少才 fallback 跨社群
}
```
